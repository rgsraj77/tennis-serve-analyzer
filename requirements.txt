# Deploy dependencies — Streamlit Community Cloud installs exactly this file.
# Keep it lean: every extra wheel is build time and a thing that can break.
# Local-only extras (FastAPI frontend, profiling) live in requirements-dev.txt.
#
# PYTHON VERSION: 3.9-3.12 only. mediapipe 0.10.14 ships no cp313 Linux wheel
# and declares no requires_python, so on 3.13+ pip attempts a source build and
# fails confusingly. Community Cloud defaults to 3.12 (fine) but offers
# 3.13/3.14 in the dropdown — do not pick them. This cannot be pinned from the
# repo; streamlit_app.py guards it at runtime.
#
# mediapipe brings opencv-contrib-python with it — do not add opencv-python
# separately or the two wheels will fight over the cv2 package.
#
# That wheel links libGL.so.1 and libgthread-2.0.so.0, which slim Linux
# containers lack, so packages.txt installs them. Two traps live in that file,
# documented here because it cannot hold comments of its own:
#   1. It must contain ONLY bare package names. Streamlit pipes it through
#      xargs into apt-get, so a "#" comment is passed as a package name
#      ("E: Unable to locate package #") and an apostrophe breaks xargs.
#   2. Use libglib2.0-0t64, NOT libglib2.0-0. Streamlit Cloud runs Debian
#      trixie, where the 64-bit time_t transition renamed the package. With
#      the old name apt falls back to a stale bullseye repo in the image and
#      dies on libffi7/libpcre3. (On bookworm or older, it is libglib2.0-0.)
mediapipe==0.10.14
numpy>=1.26,<2
scipy>=1.11
matplotlib>=3.8
imageio-ffmpeg>=0.4.9

# >=1.41: st.image() gained use_container_width there (it replaced
# use_column_width). streamlit_app.py uses it, so an older pin would install
# cleanly and then crash at render time.
streamlit>=1.41
